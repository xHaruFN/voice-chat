npm init -y npm install electron npx electron . const { app, BrowserWindow } = require('electron'); function createWindow() { const win = new BrowserWindow({ width: 300, height: 300, transparent: true, // 背景を透明にする frame: false, // 枠線をなくす alwaysOnTop: true, // 常に最前面に表示 skipTaskbar: true, // タスクバーにアイコンを表示しない webPreferences: { nodeIntegration: true, contextIsolation: false } }); // 常に最前面かつゲームを邪魔しないようにする設定 win.setAlwaysOnTop(true, 'screen-saver'); // ★重要：マウスのクリック操作を完全に透過させ、裏のゲームにそのまま伝える設定 win.setIgnoreMouseEvents(true); win.loadFile('index.html'); } app.whenReady().then(() => { createWindow(); app.on('activate', () => { if (BrowserWindow.getAllWindows().length === 0) createWindow(); }); }); app.on('window-all-closed', () => { if (process.platform !== 'darwin') app.quit(); });
